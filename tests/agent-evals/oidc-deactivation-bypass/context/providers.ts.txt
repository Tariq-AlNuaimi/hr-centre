// context: the SSO providers configured in this (synthetic) fork. Both are
// OpenID Connect providers — Auth.js reports them with account.type === "oidc",
// NOT "oauth". Supplied so the reviewer can see what the signIn guard must match.
import MicrosoftEntraID from "next-auth/providers/microsoft-entra-id";

export const providers = [
  // Entra ID is an OIDC provider (account.type === "oidc" in the signIn callback).
  MicrosoftEntraID({
    clientId: process.env.AUTH_ENTRA_ID,
    clientSecret: process.env.AUTH_ENTRA_SECRET,
  }),
  // Self-hosted Polis (BoxyHQ Jackson) bridge — also OIDC.
  {
    id: "polis",
    name: "Polis SSO",
    type: "oidc" as const,
    issuer: process.env.POLIS_ISSUER,
    clientId: process.env.POLIS_CLIENT_ID,
    clientSecret: process.env.POLIS_CLIENT_SECRET,
  },
];
